"heater_power_loss": actuators_avg.heater_pwm, heater_energy_wh:
    Rules:
    // the number of duty cycles should be porpotional to heater power
    // thus develop a "gradient" for these two variables and fail if gradient changes much

    // notice: heater_energy_wh does not change much even window.seconds fluctuate to 2, so ignore this

    gradient = (heater_energy_wh * 3600) / actuators.avg_heater_pwm
    return (gradient changes too much, e.g. 15% change from last value)
